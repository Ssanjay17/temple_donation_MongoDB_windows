# ── actual database.py source ────────────────────────────────────────────────

import os
from pymongo import MongoClient, ASCENDING
from dotenv import load_dotenv

load_dotenv()

MONGO_URI = os.getenv("MONGO_URI", "mongodb://localhost:27017")
DB_NAME   = os.getenv("DB_NAME",   "temple_donation")

client      = MongoClient(MONGO_URI, serverSelectionTimeoutMS=5_000)
db_instance = client[DB_NAME]


# ── AUTO-INCREMENT ────────────────────────────────────────────────────────────
#
#   MongoDB uses ObjectId by default.  We want integer IDs (same as MySQL) so
#   all endpoint logic continues to work without change.
#
#   Solution: a "counters" collection where each document tracks the last used
#   integer ID for one collection:
#
#       { _id: "donors", seq: 42 }
#
#   find_one_and_update + $inc is atomic — safe under concurrent requests.

def get_next_id(collection_name: str) -> int:
    """Return the next auto-increment integer ID for *collection_name*."""
    result = db_instance["counters"].find_one_and_update(
        {"_id": collection_name},
        {"$inc": {"seq": 1}},
        upsert=True,          # create the counter doc if it doesn't exist yet
        return_document=True, # return the AFTER state (new seq value)
    )
    return result["seq"]

# Usage example:
#   new_id = get_next_id("donors")   # → 1, 2, 3 … (atomically)


# ── FASTAPI DEPENDENCY ────────────────────────────────────────────────────────

def get_db():
    """Yield the MongoDB database.  Used as a FastAPI Depends parameter."""
    yield db_instance

#   @app.post("/donors")

# ── INDEXES ───────────────────────────────────────────────────────────────────
#
#   MongoDB does NOT enforce uniqueness or create indexes automatically.
#   We call create_indexes() once during app startup (@app.on_event("startup")).
#   This is idempotent — safe to call every time.

def create_indexes():
    db = db_instance

    # users
    db["users"].create_index("username",  unique=True)
    db["users"].create_index("email",     unique=True, sparse=True)

    # user_roles
    db["user_roles"].create_index("role_name", unique=True)

    # zodiac / birthstar
    db["zodiac"].create_index("zodiac_order")
    db["birthstar"].create_index(
        [("zodiac_id", ASCENDING), ("star_order", ASCENDING)]
    )

    # gotra
    db["gotra"].create_index("gotra_name", unique=True)

    # donors  ← most queried collection
    db["donors"].create_index("whatsapp_number", unique=True)
    db["donors"].create_index("email",           unique=True)
    db["donors"].create_index("first_name")

    # seva
    db["seva"].create_index("seva_name", unique=True)

    # seva_donations
    db["seva_donations"].create_index("receipt_no", unique=True)
    db["seva_donations"].create_index("donor_id")
    db["seva_donations"].create_index("seva_id")

    # seva_persons
    db["seva_persons"].create_index("seva_donation_id", unique=True)

    # seva_person_relations
    db["seva_person_relations"].create_index("seva_person_id")

    # Hindu calendar master tables
    for col in ("purnima_names", "amavasya_names"):
        db[col].create_index("name",          unique=True)
        db[col].create_index("display_order")

    for col in ("krishna_paksha_tithis", "shukla_paksha_tithis"):
        db[col].create_index("tithi_name",   unique=True)
        db[col].create_index("tithi_number")

"""
PURPOSE
───────
Provides:
  1. The `Row` class — wraps a MongoDB dict so code can use dot-access
     (user.id, user.username) instead of dict access (user["_id"]).
  2. Collection name constants (COL_USERS, COL_DONORS, …).
  3. Document builder helpers (user_doc, donor_doc, …) that create the
     correct dict shape before insertion.
  4. Thin CRUD helpers (find_one, find_all, insert_one, …) that call PyMongo
     and always return Row objects, keeping main.py clean.

WHY Row?
────────
  The old code accessed SQLAlchemy ORM objects with .attribute syntax.
  Instead of rewriting every attribute access in main.py, Row() wraps any
  MongoDB dict and provides the same dot-access.  It also maps _id → id
  automatically.
"""

# ── Row class (dot-access wrapper) ────────────────────────────────────────────

class Row:
    """
    Wraps a MongoDB document dict to allow attribute-style access.

    Examples
    --------
    doc  = db["users"].find_one({"username": "admin"})
    user = Row(doc)

    user.id          # → doc["_id"]   (maps _id → id automatically)
    user.username    # → doc["username"]
    user.is_admin    # → doc["is_admin"]
    user.missing     # → None  (never raises KeyError)
    bool(Row(None))  # → False (safe None check)
    """

    def __init__(self, doc: dict | None):
        self._doc = doc or {}

    def __bool__(self):
        return bool(self._doc)

    def __getattr__(self, name: str):
        # Map .id → ._id so callers don't need to know about MongoDB's _id
        key = "_id" if name == "id" else name
        return self._doc.get(key)

    def to_dict(self) -> dict:
        """Return the raw underlying document."""
        return self._doc


# ── CRUD helper functions ─────────────────────────────────────────────────────
#
#   All helpers return Row objects (or lists of Row objects) so main.py code
#   can use the same .attribute syntax everywhere.

def find_one(db, collection: str, query: dict) -> "Row":
    """Find a single document; returns Row (falsy if not found)."""
    return Row(db[collection].find_one(query))


def find_all(db, collection: str, query: dict = None,
             sort: list = None, limit: int = 0) -> list:
    """Find multiple documents; returns list of Row objects."""
    cursor = db[collection].find(query or {})
    if sort:
        cursor = cursor.sort(sort)
    if limit:
        cursor = cursor.limit(limit)
    return [Row(doc) for doc in cursor]


def count_docs(db, collection: str, query: dict = None) -> int:
    """Count documents matching query."""
    return db[collection].count_documents(query or {})


def insert_one(db, collection: str, doc: dict) -> int:
    """Insert document; returns the integer _id that was inserted."""
    db[collection].insert_one(doc)
    return doc["_id"]


def update_one(db, collection: str, query: dict, update: dict) -> int:
    """Update matching document; returns modified_count (0 or 1)."""
    result = db[collection].update_one(query, {"$set": update})
    return result.modified_count


def delete_one(db, collection: str, query: dict) -> int:
    """Delete one document; returns deleted_count."""
    result = db[collection].delete_one(query)
    return result.deleted_count


def delete_many(db, collection: str, query: dict) -> int:
    """Delete all matching documents; returns deleted_count."""
    result = db[collection].delete_many(query)
    return result.deleted_count


# ── Document builder helpers ──────────────────────────────────────────────────
#
#   Each function returns the correct dict shape for one collection.
#   Using builders keeps insert calls consistent and easy to audit.

def user_doc(uid: int, username: str, hashed_password: str,
             email: str = None, is_admin: int = 0,
             role_id: int = None) -> dict:
    return {
        "_id":             uid,
        "username":        username,
        "hashed_password": hashed_password,
        "email":           email,
        "is_admin":        is_admin,
        "role_id":         role_id,
    }


def user_role_doc(rid: int, role_name: str,
                  description: str = None) -> dict:
    return {
        "_id":         rid,
        "role_name":   role_name,
        "description": description,
    }


def donor_doc(did: int, **fields) -> dict:
    """
    fields may include: first_name, middle_name, last_name, gender,
    whatsapp_number, email, dob, wedding_date, zodiac_id, birthstar_id,
    gotra_id, address_line1, address_line2, city, state, pincode,
    latitude, longitude, profession_type, designation, institution,
    photo_base64, created_at
    """
    return {"_id": did, **fields}


def seva_doc(sid: int, seva_name: str, seva_description: str = None,
             default_amount_one_time: float = 0,
             default_amount_regular: float = 0,

        "default_amount_one_time":  default_amount_one_time,

GET /hindu-calendar/purnima-names  etc.

        docs = list(db["purnima_names"].find({}).sort("display_order", 1))
